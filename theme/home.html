NGEPlatform.set(new JVMAsyncPlatform());

NostrKeyPair keys = new NostrKeyPair();
NostrPool pool = new NostrPool();
NostrSubscription subscription = null;
try {
    pool.ensureRelay("wss://relay.ngengine.org").await();

    NostrFilter filter = new NostrFilter();
    filter.withKind(1);
    filter.withAuthor(keys.getPublicKey());

    subscription = pool.subscribe(filter);
    subscription.addEventListener((sub, event, stored) -> {
        System.out.println("Received: " + event.getContent());
    });
    AsyncTask.awaitAll(subscription.open());

    UnsignedNostrEvent draft = new UnsignedNostrEvent();
    draft.withKind(1);
    draft.withContent("Hello from Nostr4j!");

    NostrKeyPairSigner signer = new NostrKeyPairSigner(keys);
    SignedNostrEvent note = signer.sign(draft).await();
    pool.publish(note).await();
    System.out.println("Published: " + note.getId());

    // Keep this standalone program alive briefly to receive the note.
    Thread.sleep(5_000);
} finally {
    if (subscription != null) subscription.close();
    pool.close();
    keys.close();
}

Linux, Windows and macOS use the same Java 21+ dependencies.

          
repositories {
    mavenCentral()
{% if 'SNAPSHOT' in config.extra.versions.nostr4j or 'SNAPSHOT' in config.extra.versions.platforms %}    maven { url = uri('https://central.sonatype.com/repository/maven-snapshots') }
{% endif %}}

dependencies {
    implementation 'org.ngengine:nostr4j:{{ config.extra.versions.nostr4j }}'
    implementation 'org.ngengine:nge-platform-jvm:{{ config.extra.versions.platforms }}'
}

          
import org.ngengine.platform.NGEPlatform;
import org.ngengine.platform.jvm.JVMAsyncPlatform;

NGEPlatform.set(new JVMAsyncPlatform());

        

        

          
Use the Android adapter in your app module, with the Android SDK and Internet permission. Initialize it with your application context.

          
repositories {
    mavenCentral()
{% if 'SNAPSHOT' in config.extra.versions.nostr4j or 'SNAPSHOT' in config.extra.versions.platforms %}    maven { url = uri('https://central.sonatype.com/repository/maven-snapshots') }
{% endif %}}
dependencies {
    implementation 'org.ngengine:nostr4j:{{ config.extra.versions.nostr4j }}'
    implementation 'org.ngengine:nge-platform-android:{{ config.extra.versions.platforms }}'
}

          
import org.ngengine.platform.NGEPlatform;
import org.ngengine.platform.android.AndroidThreadedPlatform;

// In Application.onCreate(), before creating Nostr4j objects:
NGEPlatform.set(new AndroidThreadedPlatform(getApplicationContext()));
