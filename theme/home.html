NGEPlatform.set(new JVMAsyncPlatform());

NostrKeyPair keys = new NostrKeyPair();
NostrPool pool = new NostrPool();
NostrSubscription subscription = null;
try {
    pool.ensureRelay("wss://relay.ngengine.org").await();

    NostrFilter filter = new NostrFilter();
    filter.withKind(1);
    filter.withAuthor(keys.getPublicKey());

    subscription = pool.subscribe(filter);
    subscription.addEventListener((sub, event, stored) ->
        System.out.println("Received: " + event.getContent())
    );
    AsyncTask.awaitAll(subscription.open());

    UnsignedNostrEvent draft = new UnsignedNostrEvent();
    draft.withKind(1);
    draft.withContent("Hello from Nostr4J!");

    NostrKeyPairSigner signer = new NostrKeyPairSigner(keys);
    SignedNostrEvent note = signer.sign(draft).await();
    AsyncTask.awaitAll(pool.publish(note));
    System.out.println("Published: " + note.getId());

    // Keep this standalone program alive briefly to receive the note.
    Thread.sleep(5_000);
} finally {
    if (subscription != null) subscription.close();
    pool.close();
    keys.close();
}

      

    


    

      

        
Managed lifecycle
NostrPool keeps relay connections alive, reuses them by URL and restores filtered subscriptions after reconnects. Event trackers deduplicate deliveries.
Explore relay APIs →
        
Peer-to-peer RTC
Nostr signaling helps peers find each other; WebRTC data channels connect them directly, with TURN available when a direct path fails.
Explore RTC rooms →
        
RTC onion routing
Build an RTC mesh with few direct links per peer. Onion-routed messages can still reach members who are not directly connected.
See routed rooms →
        
Signers and wallets
Use local keys, a NIP-07 browser extension or a NIP-46 remote signer. Connect NIP-47 wallets for balances, invoices and payments.
Explore signing →
        
Asynchronous
Network I/O is asynchronous and exposed through AsyncTask. The JVM adapter uses virtual threads to handle many concurrent tasks without dedicating a platform thread to each one.
Explore async operations →
        
NIPs and Blossom
Helpers cover profiles, identifiers, encrypted messages, zaps and Wallet Connect. Blossom supports content-addressed media with Nostr authorization.
Browse protocol support →

Linux, Windows and macOS use the same Java 21+ dependencies.

          
repositories { mavenCentral() }

dependencies {
    implementation 'org.ngengine:nostr4j:0.3.1'
    implementation 'org.ngengine:nge-platform-jvm:0.2.4'
}

          
import org.ngengine.platform.NGEPlatform;
import org.ngengine.platform.jvm.JVMAsyncPlatform;

NGEPlatform.set(new JVMAsyncPlatform());

        

        

          
Use the Android adapter in your app module, with the Android SDK and Internet permission. Initialize it with your application context.

          
repositories { mavenCentral() }
dependencies {
    implementation 'org.ngengine:nostr4j:0.3.1'
    implementation 'org.ngengine:nge-platform-android:0.2.4'
}

          
import org.ngengine.platform.NGEPlatform;
import org.ngengine.platform.android.AndroidThreadedPlatform;

// In Application.onCreate(), before creating Nostr4J objects:
NGEPlatform.set(new AndroidThreadedPlatform(getApplicationContext()));

        

        

          
Use the NGE iOS adapter and compile your native app with the libJGLIOS toolchain on macOS with Xcode.

          
repositories { mavenCentral() }
dependencies {
    implementation 'org.ngengine:nostr4j:0.3.1'
    implementation 'org.ngengine:nge-platform-ios:0.2.4'
}

          
import org.ngengine.platform.NGEPlatform;
import org.ngengine.platform.ios.IosPlatform;

NGEPlatform.set(new IosPlatform());

Compile Java to JavaScript with TeaVM and the browser platform adapter.

          
plugins { id 'org.teavm' version '0.15.0' }
repositories {
    mavenCentral()
    maven { url = uri('https://central.sonatype.com/repository/maven-snapshots') }
}
dependencies {
    implementation 'org.ngengine:nostr4j:0.3.0-SNAPSHOT'
    implementation 'org.ngengine:nge-platform-teavm:0.3.0-SNAPSHOT'
}
teavm.js {
    mainClass = 'example.Main'
    moduleType = org.teavm.gradle.api.JSModuleType.ES2015
}

          
import org.ngengine.platform.NGEPlatform;
import org.ngengine.platform.teavm.TeaVMPlatform;

public static void main(String[] args) {
    NGEPlatform.set(new TeaVMPlatform());
    // Create Nostr4J objects after platform initialization.
}
